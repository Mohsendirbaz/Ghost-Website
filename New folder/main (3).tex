\begin{mainbox}{اطلاعات مربوط به ورود به بورس‌های بین‌المللی}
صرفاً جهت اطلاع دوستان، برخلاف تصور رایج، ارزش‌گذاری نجومی شرط ورود به بورس‌های معتبر آمریکا نیست. شرط قطعی، ارائه سه سال صورت‌های مالی حسابرسی‌شده مطابق استانداردهای حسابداری آمریکا یا بین‌المللی به‌همراه \highlight{حداقل‌هایی قابل تأمین} است.

\vspace{1em}
برای یک شرکت سودآور پنج‌ساله، کوتاه‌ترین زمان نظری مطلق از تصمیم رسمی «شروع» تا اولین روز معامله در بورس نیویورک یا نزدک حدوداً چند ماه است. این زمان‌بندی فرض می‌کند فرآیندی بی‌نقص، کاملاً مدیریت‌شده، بدون تأخیر و با فشرده‌سازی مراحل و همکاری بورس انجام شود.

\vspace{1.5em}
\begin{center}
\color{primaryBlue}\rule{0.8\textwidth}{0.8pt}
\end{center}

\vspace{0.8em}

\paragraph{الزامات واقعی بازار جهانی نزدک:}
چهار مسیر مجزا وجود دارد که حداقل‌های نظری آنها از پانزده میلیون دلار تا هفتاد و پنج میلیون دلار متغیر است. الزامات مشترک تمام مسیرها عبارتند از:

\vspace{0.5em}
\begin{itemize}[label={\textcolor{primaryBlue}{\textbullet}}]
  \item سه سال صورت‌های مالی حسابرسی‌شده مطابق استانداردهای حسابداری آمریکا یا بین‌المللی
  \item یک میلیون و یکصد هزار سهم عمومی شناور آزاد
  \item چهارصد سهامدار مستقل
  \item قیمت پیشنهادی حداقل چهار دلار
\end{itemize}

\vspace{1.5em}

\begin{pathwaybox}{pathway1}{\textbf{مسیر ۱ — مسیر درآمدی:}}
\textbf{حداقل‌های نظری:}
\begin{itemize}[label={\textcolor{accentBlue}{\textbullet}}]
  \item پانزده میلیون دلار حقوق صاحبان سهام
  \item یک میلیون دلار سود سالانه پیش از مالیات
  \item پانزده میلیون دلار ارزش بازار سهام عمومی
\end{itemize}

\textbf{مزیت کلیدی:} با اثبات سودآوری واقعی می‌توان با سرمایه نسبتاً متوسط وارد بازار شد زیرا ریسک کمتری برای سرمایه‌گذاران ایجاد می‌کند.
\end{pathwaybox}

\vspace{1em}

\begin{pathwaybox}{pathway2}{\textbf{مسیر ۲ — مسیر سرمایه:}}
\textbf{حداقل‌های نظری:}
\begin{itemize}[label={\textcolor{accentBlue}{\textbullet}}]
  \item سی میلیون دلار حقوق صاحبان سهام
  \item دو سال سابقه عملیاتی (مانع زمانی مطلق و غیرقابل کاهش)
  \item هجده میلیون دلار ارزش بازار سهام عمومی
\end{itemize}

\textbf{مزیت کلیدی:} با جذب سی میلیون دلار از طریق تعهدات و ضمانت‌های مالی می‌توان بدون نیاز به سودآوری وارد شد، اما دو سال سابقه یک الزام زمانی مطلق است.

\textbf{کاربرد بهینه:} برای شرکت‌های فناوری پیشرفته و متمرکز بر تحقیق و توسعه که درآمد کوتاه‌مدت ندارند اما پتانسیل بلندمدت بالایی دارند، ایده‌آل است.
\end{pathwaybox}

% ===================== CARBON INTENSITY MODEL =====================
\vspace{1em}
\begin{infobox}{\faCalculator ~~ مدل ریاضی شدت کربن}
\vspace{0.5em}

شدت کربن عملیاتی (gCO₂e/km) برای خودروی خودران:

\[
CI = \frac{E_{\text{drive}} + E_{\text{compute}} + E_{\text{comm}}}{\text{km}} \times \gamma_{\text{grid}} + CI_{\text{embodied}}
\]

\noindent\textbf{تأثیر ممیستور:}
\begin{itemize}[label={\textcolor{forestGreen}{\faArrowDown}}, leftmargin=*, itemsep=4pt]
    \item $E_{\text{compute}}^{\text{mem}} \approx 0.01 \times E_{\text{compute}}^{\text{GPU}}$ (بهبود 100 برابری)
    \item $E_{\text{comm}}^{\text{mem}} \approx 0.1 \times E_{\text{comm}}^{\text{cloud}}$ (بهبود 10 برابری)
    \item $\gamma_{\text{grid}}$: حساسیت زمانی (تطبیق با ساعات سبز)
\end{itemize}
\end{infobox}

افزایش کربن ناشی از محاسبات ابری در سیستم‌های خودران امروزی به چالشی اساسی تبدیل شده است. راهکار Ghost Autonomy مبتنی بر معماری EPU و سخت‌افزار مم‌رسیستور، رویکردی بنیادین برای کاهش شدت کربن (gCO₂e/km) از طریق بهینه‌سازی سه‌جانبهٔ مصرف انرژی، ارتباطات و چرخهٔ عمر سخت‌افزار ارائه می‌دهد.

\textbf{بهینه‌سازی مصرف انرژی محاسباتی:}
مم‌رسیستورها با انجام عملیات ضرب ماتریس-بردار به صورت آنالوگ درون حافظه، نیاز به جابه‌جایی داده بین حافظه و پردازنده (معضل Von Neumann) را حذف می‌کنند. این امر منجر به بهبود چشمگیر کارایی انرژی می‌شود:

\textbf{کاهش وابستگی به ارتباطات ابری:}
معماری EPU با قابلیت یادگیری آنلاین و تطبیق محلی، نیاز به آپلود داده‌های خام سنسور و به‌روزرسانی دوره‌ای مدل‌های کامل را به شدت کاهش می‌دهد. این امر منجر به صرفه‌جویی قابل توجه در انرژی ارتباطات (V2X و سلولار) می‌شود که سهم عمده‌ای در انتشار کربن دارد. بر اساس مطالعات سیستمی، حذف هر ۱ گیگابایت آپلود سلولاری معادل کاهش چندین هزار محاسبهٔ محلی از نظر انتشار CO₂ است.

\textبررسی کربن نهفته در چرخهٔ عمر:}
اگرچه تولید مم‌رسیستورها در حال حاضر شامل مواد خاص و بازدهی کمتر است (کربن نهفتهٔ بالاتر)، در سناریوهای ناوگان خودران با کاربرد بالا (مانند رباتاکسی و حمل‌ونقل باری شهری)، صرفه‌جویی عملیاتی در ۱ تا ۳ سال اول، این افزایش اولیه را جبران می‌کند. برای خودروهای شخصی با کاربرد کم، این تعادل اقتصادی‌تر و زیست‌محیطی‌تر است.

\textbf{پیامدهای سیاستی:}
این تحلیل نشان می‌دهد که استانداردهای کربن آینده باید مبتنی بر کل چرخهٔ عمر (شامل محاسبات و ارتباطات) باشند. مشوق‌های تنظیمی باید بر کاهش داده‌های ارسالی به ازای هر کیلومتر متمرکز شوند، و استقرار فناوری‌های محاسبهٔ لبه‌ای (Edge Computing) برای ناوگان تجاری در اولویت قرار گیرد.

\textbf{جمع‌بندی:}
معماری EPU مبتنی بر مم‌رسیستور، شدت کربن سیستم‌های خودران را از طریق کاهش همزمان انرژی محاسباتی و ارتباطی به حداقل می‌رساند، و مسیری عملی برای تحقق هدف «صفر خالص» در صنعت خودروهای خودران ارائه می‌دهد.

\begin{securitybox}{امنیت عمومی: تضمین قابلیت اطمینان و ایمنی در سیستم‌های خودران}

\textbf{چالش امنیت عمومی در خودروهای خودران:}
سیستم‌های خودران به عنوان بخشی از زیرساخت حیاتی جامعه، باید در برابر حملات سایبری، خطاهای سخت‌افزاری، و شرایط غیرمنتظره مقاوم باشند. معماری Ghost Autonomy با طراحی چندلایه، امنیت عمومی را در سه سطح تضمین می‌کند.

\textbf{سطح ۱: تحمل خطای بیزانسی توزیع‌شده}
معماری Queen Bee (ملکه زنبور) با پروتکل اجماع ۳-مرحله‌ای بیزانسی، امکان تحمل تا ⌊(n-1)/3⌋ گره خرابکار یا معیوب را فراهم می‌آورد. این پروتکل شامل:
\begin{itemize}[label=\textcolor{warningRed}{\textbullet}]
  \item فاز Broadcast: انتشار تصمیم اولیه
  \item فاز Echo: تأیید تصمیم توسط گره‌ها
  \item فاز Ready: نهایی‌سازی اجماع
\end{itemize}
با تأخیر کل کمتر از ۱.۵ میکروثانیه، این سیستم حتی در حضور حملات هماهنگ نیز تصمیمات ایمن تولید می‌کند.

\textbf{سطح ۲: معماری ایمنی غیرقابل رمزگذاری/مسیریابی/مجوز}

\textbf{سطح ۳: انطباق با استانداردهای ایمنی عملکردی}
معماری EPU به گونه‌ای طراحی شده است که الزامات ASIL-D استاندارد ISO 26262 را برآورده می‌کند، شامل:
\begin{itemize}[label=\textcolor{warningRed}{\textbullet}]
  \item اعتبارسنجی مداوم از طریق تست‌های تزریق خطا
  \item مکانیزم‌های بازیابی خودکار (Safety Rollback)
  \item لاگ‌های امنیتی Write-Ahead با قابلیت ممیزی کامل
\end{itemize}

\textbf{حفاظت از حریم خصوصی داده‌ها:}
با کاهش وابستگی به ابر و پردازش محلی داده‌های حساس (مانند تصاویر دوربین‌ها)، معماری EPU خطر نشت اطلاعات شخصی را به حداقل می‌رساند. مدل‌های یادگیری آنلاین تنها بر روی ویژگی‌های انتزاعی (Moments) کار می‌کنند، نه داده‌های خام.

\textbf{نتیجه‌گیری:}
رویکرد چندلایهٔ Ghost Autonomy—ترکیب تحمل خطای بیزانسی، اصول طراحی امنیتی سخت‌افزار-محور، و انطباق با استانداردهای صنعتی—چارچوبی جامع برای تضمین امنیت عمومی در عصر خودروهای خودران ارائه می‌دهد.
